local function QuickTween(object, info, properties)
    local tween = TweenService:Create(object, info or TweenInfo.new(0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), properties)
    tween:Play()
    return tween
end

    --// Dragging System
    local function makeDraggable(guiObject, dragTarget)
        dragTarget = dragTarget or guiObject
        local dragging, dragStart, startPos

        dragTarget.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = guiObject.Position
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                    end
                end)
            end
        end)

        UserInputService.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                if dragging then
                    local delta = input.Position - dragStart
                    guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
                end
            end
        end)
    end

        QuickTween(afkDot, nil, {Position = UDim2.new(1, -17, 0.5, -7), BackgroundColor3 = Color3.fromRGB(255, 255, 255)})

        pcall(function()

                VirtualUser:CaptureController()
                VirtualUser:ClickButton2(Vector2.new(0,0))
            end)
        end)

        -- Periodic Mouse Movement Loop (Fallback protection)
        antiAfkLoop = task.spawn(function()
            while afkState do
                task.wait(15)
                pcall(function()
                    VirtualUser:CaptureController()
                    VirtualUser:MouseMove(Vector2.new(math.random(-10, 10), math.random(-10, 10)))
                end)
            end
        end)
    else
        QuickTween(afkTogglePill, nil, {BackgroundColor3 = Color3.fromRGB(35, 35, 48)})
        QuickTween(afkDot, nil, {Position = UDim2.new(0, 3, 0.5, -7), BackgroundColor3 = Theme.TextSecondary})
        
        if antiAfkConnection then
            antiAfkConnection:Disconnect()
            antiAfkConnection = nil
        end
        if antiAfkLoop then
            task.cancel(antiAfkLoop)
            antiAfkLoop = nil
        end
    end
end

            refreshStroke.Color = Theme.Stroke
            refreshStroke.Thickness = 1

            refreshBtn.MouseEnter:Connect(function() QuickTween(refreshBtn, nil, {BackgroundColor3 = Theme.CardHover}) end)
            refreshBtn.MouseLeave:Connect(function() QuickTween(refreshBtn, nil, {BackgroundColor3 = Color3.fromRGB(16, 16, 22)}) end)

            local dropdownList = Instance.new("ScrollingFrame")
            dropdownList.Parent = container
            dropdownList.Active = true
            dropdownList.ScrollingEnabled = true
            dropdownList.BackgroundTransparency = 1
            dropdownList.Position = UDim2.new(0, 0, 0, 40)
            dropdownList.Size = UDim2.new(1, 0, 0, math.min(#options * 30, 120))
            dropdownList.CanvasSize = UDim2.new(0, 0, 0, #options * 30)
            dropdownList.ScrollBarThickness = 2

            local listLayout = Instance.new("UIListLayout")
            listLayout.Parent = dropdownList
            listLayout.SortOrder = Enum.SortOrder.LayoutOrder
            listLayout.Padding = UDim.new(0, 3)

            local DropdownObject = {}

            function DropdownObject:Refresh(newOptions, keepSelection)
                options = newOptions or {}
                
                -- Clean existing elements
                for _, child in ipairs(dropdownList:GetChildren()) do
                    if child:IsA("TextButton") then
                        child:Destroy()
                    end
                end

                if not keepSelection or not table.find(options, selectedOption) then
                    selectedOption = options[1] or ""
                    selectBtn.Text = "  " .. text .. ": " .. tostring(selectedOption)
                end

                for _, opt in ipairs(options) do
                    local optBtn = Instance.new("TextButton")
                    optBtn.Parent = dropdownList
                    optBtn.BackgroundColor3 = Color3.fromRGB(16, 16, 22)
                    optBtn.Size = UDim2.new(1, -16, 0, 26)
                    optBtn.Position = UDim2.new(0, 8, 0, 0)
                    optBtn.Font = Enum.Font.Gotham
                    optBtn.Text = tostring(opt)
                    optBtn.TextColor3 = Theme.TextSecondary
                    optBtn.TextSize = 10

                    local optCorner = Instance.new("UICorner")
                    optCorner.CornerRadius = UDim.new(0, 6)
                    optCorner.Parent = optBtn

                    optBtn.MouseEnter:Connect(function() QuickTween(optBtn, nil, {BackgroundColor3 = Theme.CardHover, TextColor3 = Theme.TextPrimary}) end)
                    optBtn.MouseLeave:Connect(function() QuickTween(optBtn, nil, {BackgroundColor3 = Color3.fromRGB(16, 16, 22), TextColor3 = Theme.TextSecondary}) end)

                    optBtn.Activated:Connect(function()
                        selectedOption = opt
                        selectBtn.Text = "  " .. text .. ": " .. tostring(selectedOption)
                        isOpen = false
                        QuickTween(arrowIcon, nil, {Rotation = 0})
                        QuickTween(container, nil, {Size = UDim2.new(1, -8, 0, 38)})
                        pcall(function() callback(selectedOption) end)
                    end)
                end

                local listHeight = math.min(#options * 30, 120)
                dropdownList.Size = UDim2.new(1, 0, 0, listHeight)
                dropdownList.CanvasSize = UDim2.new(0, 0, 0, #options * 30)

                if isOpen then
                    QuickTween(container, nil, {Size = UDim2.new(1, -8, 0, 44 + listHeight)})
                end
            end

            refreshBtn.Activated:Connect(function()
                QuickTween(refreshBtn, TweenInfo.new(0.2), {Rotation = refreshBtn.Rotation + 360})
                if onRefresh then
                    local updated = onRefresh()
                    if type(updated) == "table" then
                        DropdownObject:Refresh(updated, true)
                    end
                end
            end)

            selectBtn.Activated:Connect(function()
                isOpen = not isOpen
                local listHeight = math.min(#options * 30, 120)
                local targetHeight = isOpen and (44 + listHeight) or 38
                QuickTween(arrowIcon, nil, {Rotation = isOpen and 180 or 0})
                QuickTween(container, nil, {Size = UDim2.new(1, -8, 0, targetHeight)})
            end)

            DropdownObject:Refresh(options, true)
            return DropdownObject
        end

        return TabSection
    end

    return Window
end

return Library
